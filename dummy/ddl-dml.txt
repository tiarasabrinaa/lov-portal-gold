-- ============================================================
-- lov_gold LAYER - EMPLOYER DOMAIN (BigQuery Standard SQL)
-- Dataset: lov_gold
--
-- Struktur:
--   lov_gold.employer              -> tabel utama (wide/flat), 1 row per employer
--   lov_gold.employer_address      -> semua address (termasuk primary), flag is_primary
--   lov_gold.employer_contact      -> semua contact (termasuk primary), flag is_primary
--   lov_gold.employer_account      -> semua account per employer
--   lov_gold.employer_shareholder  -> semua shareholder per employer
--
-- Catatan: BigQuery PK/FK constraint sifatnya NOT ENFORCED
-- (cuma metadata utk query optimizer & dokumentasi lineage,
--  gak ada validasi otomatis kayak RDBMS biasa). Makanya urutan
--  insert di bawah tetap dijaga manual: employer dulu, baru child table.
-- ============================================================


-- ============================================================
-- SET EXPIRATION - buat tabel lov_gold (jaga-jaga kalau lupa didrop manual)
-- Set expired 7 hari dari sekarang, tinggal ganti INTERVAL sesuai kebutuhan
-- ============================================================
ALTER TABLE `lov_gold.employer`             SET OPTIONS (expiration_timestamp = TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 7 DAY));
ALTER TABLE `lov_gold.employer_address`     SET OPTIONS (expiration_timestamp = TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 7 DAY));
ALTER TABLE `lov_gold.employer_contact`     SET OPTIONS (expiration_timestamp = TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 7 DAY));
ALTER TABLE `lov_gold.employer_account`     SET OPTIONS (expiration_timestamp = TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 7 DAY));
ALTER TABLE `lov_gold.employer_shareholder` SET OPTIONS (expiration_timestamp = TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 7 DAY));

-- ========================================

-- ------------------------------------------------------------
-- 1. lov_gold.employer (tabel utama)
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer` (
  employer_id                  STRING  NOT NULL,
  group_id                     STRING,
  nob_id                       STRING,
  cif                          STRING  NOT NULL,
  stock_code                   STRING,
  employer_name                STRING  NOT NULL,
  ipo_date                     DATE,
  is_top_1000                  BOOL,
  is_pks                       BOOL,
  tier                         STRING,
  group_name                   STRING,
  cl_account                   STRING,
  cl_loan_type                 STRING,
  primary_address_detail       STRING,
  primary_address_subdistrict  STRING,
  primary_address_district     STRING,
  primary_address_city         STRING,
  primary_address_province     STRING,
  primary_address_postcode     STRING,
  primary_contact              STRING,
  shareholder_count            INT64,
  top_shareholder_name         STRING,
  top_shareholder_pct          NUMERIC,
  account_count                INT64,
  primary_account_number       STRING,
  snapshot_date                DATE    NOT NULL,
  PRIMARY KEY (employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 2. lov_gold.employer_address (all address, ada is_primary)
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer_address` (
  address_id      STRING  NOT NULL,
  employer_id     STRING  NOT NULL,
  address_type    STRING,
  address_detail  STRING,
  subdistrict     STRING,
  district        STRING,
  city            STRING,
  province        STRING,
  postcode        STRING,
  is_primary      BOOL,
  is_current      BOOL,
  effective_date  DATE,
  expiry_date     DATE,
  PRIMARY KEY (address_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 3. lov_gold.employer_contact (all contact, ada is_primary)
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer_contact` (
  contact_id     STRING  NOT NULL,
  employer_id    STRING  NOT NULL,
  contact_type   STRING,
  contact_value  STRING,
  is_primary     BOOL,
  snapshot_date  DATE,
  PRIMARY KEY (contact_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 4. lov_gold.employer_account
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer_account` (
  account_id      STRING  NOT NULL,
  employer_id     STRING  NOT NULL,
  account_number  STRING,
  account_type    STRING,
  snapshot_date   DATE,
  PRIMARY KEY (account_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 5. lov_gold.employer_shareholder
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer_shareholder` (
  shareholder_id          STRING  NOT NULL,
  employer_id             STRING  NOT NULL,
  shareholder_name        STRING,
  shareholder_amount      NUMERIC,
  shareholder_percentage  NUMERIC,
  shareholder_type        STRING,
  snapshot_date           DATE,
  PRIMARY KEY (shareholder_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- ============================================================
-- VERIFIKASI - pastiin FK "jalan" (BigQuery gak enforce FK,
-- jadi validnya dicek manual pakai query, bukan pas insert)
-- ============================================================

-- 1. Cek ada child row yang employer_id-nya orphan (harusnya 0 row)
SELECT 'employer_address' AS tabel, a.employer_id
FROM `lov_gold.employer_address` a
LEFT JOIN `lov_gold.employer` e ON a.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'employer_contact', c.employer_id
FROM `lov_gold.employer_contact` c
LEFT JOIN `lov_gold.employer` e ON c.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'employer_account', acc.employer_id
FROM `lov_gold.employer_account` acc
LEFT JOIN `lov_gold.employer` e ON acc.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'employer_shareholder', s.employer_id
FROM `lov_gold.employer_shareholder` s
LEFT JOIN `lov_gold.employer` e ON s.employer_id = e.employer_id
WHERE e.employer_id IS NULL;

-- 2. Cek shareholder_count & account_count di lov_gold.employer
--    match sama jumlah row aktual di child table
SELECT
  e.employer_id,
  e.shareholder_count AS shareholder_count_flat,
  COUNT(s.shareholder_id) AS shareholder_count_actual,
  e.account_count AS account_count_flat,
  COUNT(DISTINCT acc.account_id) AS account_count_actual
FROM `lov_gold.employer` e
LEFT JOIN `lov_gold.employer_shareholder` s ON e.employer_id = s.employer_id
LEFT JOIN `lov_gold.employer_account` acc ON e.employer_id = acc.employer_id
GROUP BY e.employer_id, e.shareholder_count, e.account_count
ORDER BY e.employer_id;

-- 3. Contoh query "self-service" tanpa join sama sekali
--    (ini yang mau dibuktiin ke SPV: cukup 1 tabel buat kasus umum)
SELECT employer_name, tier, primary_address_city, primary_contact, top_shareholder_name
FROM `lov_gold.employer`
WHERE is_top_1000 = TRUE;

-- 4. Contoh query drill-down (baru butuh join, buat kasus non-umum)
SELECT e.employer_name, a.address_type, a.address_detail, a.city
FROM `lov_gold.employer` e
JOIN `lov_gold.employer_address` a ON e.employer_id = a.employer_id
ORDER BY e.employer_id, a.is_primary DESC;

-- 1. Cek orphan nob_id & postcode_id (harusnya 0 row)
SELECT 'employer.nob_id' AS kolom, e.employer_id, e.nob_id
FROM `lov_gold.employer` e
LEFT JOIN `lov_gold.nob` n ON e.nob_id = n.nob_id
WHERE e.nob_id IS NOT NULL AND n.nob_id IS NULL
UNION ALL
SELECT 'employer_address.postcode_id', a.address_id, a.postcode_id
FROM `lov_gold.employer_address` a
LEFT JOIN `lov_gold.postcode` p ON a.postcode_id = p.post_code_id
WHERE a.postcode_id IS NOT NULL AND p.post_code_id IS NULL;

-- 2. Contoh query gabungan employer + sektor bisnis + kode wilayah lengkap
SELECT
  e.employer_name,
  n.original_sector,
  n.original_industry,
  n.kbli,
  a.address_detail,
  p.kode_pos,
  p.kode_kemendagri
FROM `lov_gold.employer` e
JOIN `lov_gold.nob` n ON e.nob_id = n.nob_id
JOIN `lov_gold.employer_address` a ON e.employer_id = a.employer_id AND a.is_primary = TRUE
JOIN `lov_gold.postcode` p ON a.postcode_id = p.post_code_id
ORDER BY e.employer_id;
