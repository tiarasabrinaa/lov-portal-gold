-- ============================================================
-- COMPANY lov_GOLD (v3 / "atwork" design) - BigQuery Standard SQL
-- Dataset: lov_gold
-- 4 tabel: gold_employer_profile, gold_employer_account,
--          gold_stakeholder, gold_address
--
-- Sumber: dummy/design-gold-company-atwork.png (ERD, versi terbaru
-- yang di-upload 2026-07-27). Ini MENGGANTIKAN desain v2
-- (dummy/ddl-dml-company-atwork.txt: employer / employer_account /
-- branch / address). Nama tabel sengaja beda (prefix gold_) supaya
-- tidak bentrok dengan tabel v2/v1 lama - boleh hidup bareng sampai
-- tabel lama di-drop manual.
--
-- Perubahan utama dari v2:
-- - branch jadi standalone table di v2 -> sekarang field branch_*
--   didenormalisasi langsung ke gold_employer_account (ga ada lagi
--   tabel branch terpisah).
-- - cif sekarang ada di SEMUA tabel (bukan cuma employer & yang
--   join lewat employer_id) - jadi cif dipakai sebagai lookup key
--   utama di API, employer_id/FK lain tetap disimpen sebagai kolom
--   deskriptif/referensi tapi API-nya standar ke by-cif.
-- - stakeholder (gold_stakeholder) baru ada di desain ini - belum
--   ada sama sekali di v2.
--
-- Catatan asumsi (ga eksplisit kelihatan di gambar ERD, cek balik):
-- - snapshot_date ditambahin di gold_employer_account & gold_address
--   walau di gambar 2 tabel ini ga kelihatan ada snapshot_date -
--   diasumsikan konsisten sama pola gold layer yang lain (employer
--   profile & stakeholder eksplisit ada snapshot_date di gambar).
-- - Gambar nunjukin cif di gold_employer_account sebagai FK ke
--   gold_employer_profile, tapi BigQuery cuma izinin FK yang match
--   persis sama PRIMARY KEY tabel tujuan (employer_id, bukan cif) -
--   jadi cif di gold_employer_account dibiarin kolom deskriptif aja
--   tanpa constraint FK formal (sama pola kayak v2 dulu).
--
-- Update 2026-07-27: gold_employer_account awalnya cuma punya cif
-- (ga ada employer_id sama sekali, lihat catatan di atas). Ditambahin
-- kolom employer_id (backfill dari cif via join ke gold_employer_profile)
-- supaya endpoint /accounts/by-employer-id/{employer_id} bisa query
-- langsung tanpa resolve cif dulu, dan FK formal employer_id->employer_id
-- sekarang valid (match PRIMARY KEY gold_employer_profile).
--
-- Update 2026-07-28: fix bug di gold_stakeholder DDL (kolom group_id
-- tadinya ga diakhiri koma -> syntax error, dan group_name kepasang
-- 2x -> duplicate column error). DML disamain ke DDL yang ada: kolom
-- sector/sub_sector/subindustry yang sempat muncul di draft DML
-- dihapus (DDL cuma punya 1 kolom klasifikasi industri: `industry`),
-- data dummy jadi masuk ke kolom industry itu aja.
--
-- Update 2026-07-28 (lanjutan) - app/api & app/schemas disinkronin ke
-- skema final ini:
-- - gold_employer_profile: sector/sub_sector/subindustry DIHAPUS dari
--   kode (udah ga ada di DDL), tiering_code & tiering_label DITAMBAH
--   (masih NULL semua di DML, belum ada sumber data buat backfill).
-- - gold_employer_account: employer_name/group_id/group_name
--   DITAMBAH ke kode (didenormalisasi, tapi DML belum ngisi value-nya
--   jadi masih NULL - PR selanjutnya kalau mau di-backfill dari join
--   ke gold_employer_profile by employer_id).
-- - gold_stakeholder: sector DIHAPUS dari kode, diganti industry
--   (data lama sector udah dipindah manual ke industry di DML),
--   group_id DITAMBAH (masih NULL, belum di-backfill).
-- - gold_address: postcode DITAMBAH (masih NULL, DML cuma ngisi
--   postcode_id).
-- - Ketemu bug URL collision: endpoint by-employer-id di company.py
--   sempat dipakein path /employer/{employer_id} (single segment),
--   yang BENTROK sama /employer/{cif} di company_cif.py (sama-sama
--   1 segment dinamis di bawah /employer). company_router didaftar
--   duluan di router.py jadi selalu menang -> lookup by-cif via
--   /employer/{cif} diam-diam kepanggil sebagai by-employer_id dan
--   selalu kosong. Fix: path by-employer-id dibalikin pakai prefix
--   eksplisit /employer/by-employer-id/{employer_id} (juga buat
--   account/stakeholder/address), /employer/{cif} dkk tetap plain.
-- ============================================================

CREATE SCHEMA IF NOT EXISTS `lov_gold`;

-- ------------------------------------------------------------
-- 1. gold_employer_profile
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.gold_employer_profile` (
  employer_id          STRING  NOT NULL,
  group_id             STRING,
  nob_id               STRING,
  cif                  STRING  NOT NULL,
  employer_name        STRING  NOT NULL,
  employer_code        STRING,
  industry             STRING,
  tiering_code         STRING,
  tiering_label        STRING,
  group_name           STRING,
  primary_email        STRING,
  primary_contact_no   STRING,
  snapshot_date        DATE    NOT NULL,
  PRIMARY KEY (employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 2. gold_employer_account (branch_* didenormalisasi di sini)
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.gold_employer_account` (
  account_id        STRING  NOT NULL,
  cif               STRING,
  employer_id       STRING,
  employer_name     STRING,
  group_id          STRING,
  group_name        STRING,
  branch_code       STRING,
  branch_name       STRING,
  branch_address1   STRING,
  branch_address2   STRING,
  branch_telephone  STRING,
  branch_email      STRING,
  account_number    STRING,
  currency          STRING,
  product_code      STRING,
  account_type      STRING,   -- is_giro: 'GIRO' | 'DEPOSITO' | dst.
  is_sharia         BOOL,
  snapshot_date     DATE    NOT NULL,
  PRIMARY KEY (account_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.gold_employer_profile`(employer_id) NOT ENFORCED
  -- employer_id ditambahin belakangan (awalnya cuma ada cif di gambar
  -- ERD) supaya lookup langsung by-employer_id ga perlu resolve lewat
  -- cif dulu. cif tetap dibiarin sebagai kolom deskriptif tanpa FK
  -- formal (BigQuery mewajibkan kolom FK match persis PRIMARY KEY
  -- tabel tujuan, dan cif bukan PK gold_employer_profile).
);

-- ------------------------------------------------------------
-- 3. gold_stakeholder
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.gold_stakeholder` (
  stakeholder_id        STRING  NOT NULL,
  cif                   STRING,
  employer_id           STRING  NOT NULL,
  employer_name         STRING,
  group_id              STRING,
  group_name            STRING,
  stock_code            STRING,
  industry              STRING,
  shareholder_type      STRING,
  stakeholder_name      STRING,
  ownership_amount      NUMERIC,
  ownership_percentage  NUMERIC,
  designation           STRING,
  snapshot_date         DATE    NOT NULL,
  PRIMARY KEY (stakeholder_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.gold_employer_profile`(employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 4. gold_address
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.gold_address` (
  address_id      STRING  NOT NULL,
  employer_id     STRING  NOT NULL,
  cif             STRING,
  postcode_id     STRING,
  postcode        STRING,
  is_primary      BOOL,
  rt              STRING,
  rw              STRING,
  address_detail  STRING,
  subdistrict     STRING,
  district        STRING,
  city            STRING,
  province        STRING,
  snapshot_date   DATE    NOT NULL,
  PRIMARY KEY (address_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.gold_employer_profile`(employer_id) NOT ENFORCED
);

-- VERIFIKASI
-- ============================================================

-- 1. Orphan check cif / employer_id (harusnya 0 row)
SELECT 'gold_employer_account' AS tabel, a.cif
FROM `lov_gold.gold_employer_account` a
LEFT JOIN `lov_gold.gold_employer_profile` e ON a.cif = e.cif
WHERE e.cif IS NULL
UNION ALL
SELECT 'gold_stakeholder', s.employer_id
FROM `lov_gold.gold_stakeholder` s
LEFT JOIN `lov_gold.gold_employer_profile` e ON s.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'gold_address', ad.employer_id
FROM `lov_gold.gold_address` ad
LEFT JOIN `lov_gold.gold_employer_profile` e ON ad.employer_id = e.employer_id
WHERE e.employer_id IS NULL;

-- 2. Setiap employer harus punya tepat 1 alamat primary
SELECT employer_id, COUNT(*) AS primary_count
FROM `lov_gold.gold_address`
WHERE is_primary = TRUE
GROUP BY employer_id
HAVING COUNT(*) != 1;

-- 3. Contoh query self-service tanpa join (semua entity punya cif sendiri)
SELECT employer_name, industry, primary_email, cif
FROM `lov_gold.gold_employer_profile`;

SELECT account_number, branch_name, account_type, cif
FROM `lov_gold.gold_employer_account`
WHERE cif = '1000000001';
