-- ============================================================
-- NOB (NATURE OF BUSINESS) lov_GOLD - BigQuery Standard SQL
-- Dataset: lov_gold
-- 1 tabel: nob
--
-- Catatan:
-- - Kolom persis ngikutin app/schemas/nature_of_business.py
--   (NatureOfBusinessRead): nob_id, master_code,
--   master_description, source_system, original_code,
--   original_description, snapshot_date.
-- - nob_id itu identitas si business-nya sendiri, GAK ada
--   hubungannya sama dari source_system mana asalnya. Jadi 1
--   nob_id yang SAMA dipakai ulang di beberapa baris, satu per
--   source_system (SIBS, ASCEND), karena tiap sistem asal punya
--   istilah/kode sendiri buat business yang sama. Contoh: nob_id
--   NOB01 ("Perbankan Umum") muncul 2x:
--     * SIBS   -> original_description "Finansial"
--     * ASCEND -> original_description "Perbankan & Jasa Keuangan Korporat"
--   Pola ini diulang buat semua 5 nob_id di bawah. Makanya PRIMARY
--   KEY tabel ini gabungan (nob_id, source_system), bukan nob_id
--   doang.
-- - app/api/endpoints/nature_of_business.py punya 2 endpoint:
--     * GET /nature-of-business/          -> get all, gak ada filter
--     * GET /nature-of-business/{nob_id}   -> get by id, tanpa filter
--       balikin semua source_system buat id itu; plus query param
--       opsional ?source_system= buat narrow ke satu, contoh:
--         GET /nature-of-business/NOB01               -> 2 baris (SIBS + ASCEND)
--         GET /nature-of-business/NOB01?source_system=SIBS -> 1 baris
-- - nob_id NOB01..NOB05 sengaja disamain sama nob_id yang udah
--   dipakai di `lov_gold.view_employer` (lihat ddl-dml.txt), biar
--   relasinya tetep konsisten kalau dua-duanya di-load.
-- ============================================================

CREATE SCHEMA IF NOT EXISTS `lov_gold`;

-- ------------------------------------------------------------
-- 1. nob
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.nob` (
  nob_id                STRING  NOT NULL,
  master_code           STRING,
  master_description    STRING,
  source_system         STRING,
  original_code         STRING,
  original_description  STRING,
  snapshot_date         DATE    NOT NULL,
  PRIMARY KEY (nob_id, source_system) NOT ENFORCED
);

-- VERIFIKASI

-- 1. Cek tiap nob_id punya 2 source_system (harus semua = 2)
SELECT nob_id, COUNT(DISTINCT source_system) AS total_source_system
FROM `lov_gold.nob`
GROUP BY nob_id
ORDER BY nob_id;

-- 2. Orphan check ke view_employer.nob_id (kalau ddl-dml.txt view_employer
--    udah di-load duluan; harusnya 0 row karena NOB01..NOB05 dipakai di sana)
SELECT e.nob_id
FROM `lov_gold.view_employer` e
LEFT JOIN `lov_gold.nob` n ON e.nob_id = n.nob_id
WHERE n.nob_id IS NULL;

-- 3. Simulasi GET /nature-of-business/ (get all)
SELECT nob_id, master_code, master_description, source_system, original_code, original_description, snapshot_date
FROM `lov_gold.nob`
ORDER BY master_code;

-- 4. Simulasi GET /nature-of-business/NOB01 (get by id, tanpa filter source -> 2 baris)
SELECT nob_id, master_code, master_description, source_system, original_code, original_description, snapshot_date
FROM `lov_gold.nob`
WHERE nob_id = 'NOB01'
ORDER BY master_code;

-- 5. Simulasi GET /nature-of-business/NOB01?source_system=SIBS (get by id + filter source -> 1 baris)
SELECT nob_id, master_code, master_description, source_system, original_code, original_description, snapshot_date
FROM `lov_gold.nob`
WHERE nob_id = 'NOB01' AND source_system = 'SIBS'
ORDER BY master_code;
