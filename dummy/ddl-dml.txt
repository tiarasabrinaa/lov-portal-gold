-- =====================================================================
-- LOV GOLD LAYER — Dummy DDL & DML (BigQuery)
-- Dataset: lov_gold
-- Purpose: sandbox / demo schema for LoV Portal Gold-layer company
--          master data (COMP, COMP_ADDR, COMP_GROUP, COMP_SHRLDR,
--          COMP_CONT, KBLI_IDX, POSTCODE)
-- Note: Gold layer is presented as current-state (no SCD2 here —
--       SCD2 history lives in Silver, per your architecture decision).
-- =====================================================================

-- Replace `your_project` with your actual GCP project id before running,
-- or drop the project prefix entirely if you run this with a default
-- project already set in your BigQuery session.

CREATE SCHEMA IF NOT EXISTS `your_project.lov_gold`
OPTIONS (
  location = 'asia-southeast2', -- Jakarta region
  description = 'Gold layer: curated LoV (List of Values) company master data',
  -- Not required anymore now that billing is enabled (this was the
  -- sandbox-mode requirement), but kept here as good hygiene since
  -- this is dummy/testing data — tables auto-expire instead of
  -- lingering and quietly costing storage.
  -- Set to NULL (remove these two lines) if you want this to become
  -- permanent/production data with no auto-expiry.
  default_table_expiration_days = 30,
  default_partition_expiration_days = 30
);

-- =====================================================================
-- 1. COMP — Company master
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.comp` (
  comp_id           STRING       NOT NULL OPTIONS(description = 'Unique company identifier (surrogate key)'),
  comp_code         STRING       OPTIONS(description = 'Business/source company code'),
  comp_name         STRING       NOT NULL,
  comp_short_name   STRING,
  comp_type         STRING       OPTIONS(description = 'e.g. PT, CV, FIRMA, KOPERASI'),
  npwp              STRING       OPTIONS(description = 'Tax ID, 15/16 digit'),
  nib               STRING       OPTIONS(description = 'Nomor Induk Berusaha'),
  kbli_code         STRING       OPTIONS(description = 'FK to kbli_idx.kbli_code'),
  industry_sector   STRING,
  incorporation_date DATE,
  comp_status       STRING       OPTIONS(description = 'ACTIVE, INACTIVE, DISSOLVED'),
  source_system     STRING       OPTIONS(description = 'e.g. MADAM, AHU, MANUAL'),
  gold_load_ts      TIMESTAMP    NOT NULL OPTIONS(description = 'ETL load timestamp into Gold')
)
CLUSTER BY comp_id
OPTIONS (
  description = 'Gold: current-state company master (one row per company)'
);

-- =====================================================================
-- 2. COMP_ADDR — Company address
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.comp_addr` (
  addr_id           STRING       NOT NULL,
  comp_id           STRING       NOT NULL OPTIONS(description = 'FK to comp.comp_id'),
  addr_type         STRING       OPTIONS(description = 'REGISTERED, DOMICILE, BRANCH, MAILING'),
  address_line      STRING,
  rt_rw             STRING,
  kelurahan         STRING,
  kecamatan         STRING,
  kabupaten_kota    STRING,
  provinsi          STRING,
  postal_code       STRING       OPTIONS(description = 'FK to postcode.postal_code'),
  latitude          FLOAT64,
  longitude         FLOAT64,
  is_primary        BOOL,
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY comp_id
OPTIONS (
  description = 'Gold: current company addresses, one-to-many with comp'
);

-- =====================================================================
-- 3. COMP_GROUP — Corporate group / holding structure
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.comp_group` (
  group_id          STRING       NOT NULL,
  comp_id           STRING       NOT NULL OPTIONS(description = 'FK to comp.comp_id (member company)'),
  parent_comp_id    STRING       OPTIONS(description = 'FK to comp.comp_id (holding/parent company)'),
  group_name        STRING,
  relationship_type STRING       OPTIONS(description = 'SUBSIDIARY, AFFILIATE, HOLDING'),
  ownership_level   INT64        OPTIONS(description = 'Depth in group hierarchy'),
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY comp_id
OPTIONS (
  description = 'Gold: corporate group / ownership hierarchy'
);

-- =====================================================================
-- 4. COMP_SHRLDR — Shareholders
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.comp_shrldr` (
  shrldr_id         STRING       NOT NULL,
  comp_id           STRING       NOT NULL OPTIONS(description = 'FK to comp.comp_id'),
  shareholder_name  STRING       NOT NULL,
  shareholder_type  STRING       OPTIONS(description = 'INDIVIDUAL, CORPORATE, GOVERNMENT'),
  identity_no       STRING       OPTIONS(description = 'KTP or corporate NPWP'),
  ownership_pct     NUMERIC(5,2) OPTIONS(description = 'Ownership percentage, 0-100'),
  shares_count      INT64,
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY comp_id
OPTIONS (
  description = 'Gold: shareholder composition per company'
);

-- =====================================================================
-- 5. COMP_CONT — Company contact points
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.comp_cont` (
  cont_id           STRING       NOT NULL,
  comp_id           STRING       NOT NULL OPTIONS(description = 'FK to comp.comp_id'),
  contact_type      STRING       OPTIONS(description = 'PHONE, EMAIL, FAX, WEBSITE, PIC'),
  contact_value     STRING,
  pic_name          STRING       OPTIONS(description = 'Person in charge, if contact_type = PIC'),
  is_primary        BOOL,
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY comp_id
OPTIONS (
  description = 'Gold: company contact points'
);

-- =====================================================================
-- 6. KBLI_IDX — KBLI (Indonesian Standard Industrial Classification) index
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.kbli_idx` (
  kbli_code         STRING       NOT NULL,
  kbli_desc         STRING       NOT NULL,
  kbli_category     STRING       OPTIONS(description = 'e.g. A, B, C section code'),
  kbli_category_desc STRING,
  risk_level        STRING       OPTIONS(description = 'LOW, MEDIUM, HIGH per OSS-RBA'),
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY kbli_code
OPTIONS (
  description = 'Gold: KBLI reference / lookup table'
);

-- =====================================================================
-- 7. POSTCODE — Wilayah / postal code reference
-- =====================================================================
CREATE OR REPLACE TABLE `your_project.lov_gold.postcode` (
  postal_code       STRING       NOT NULL,
  kelurahan         STRING,
  kecamatan         STRING,
  kabupaten_kota    STRING,
  provinsi          STRING,
  gold_load_ts      TIMESTAMP    NOT NULL
)
CLUSTER BY postal_code
OPTIONS (
  description = 'Gold: wilayah / postal code reference (used for address matching)'
);
