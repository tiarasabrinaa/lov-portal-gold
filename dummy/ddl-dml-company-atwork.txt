-- ============================================================
-- COMPANY lov_GOLD (v2 / "atwork" design) - BigQuery Standard SQL
-- Dataset: lov_gold
-- 4 tabel: employer, employer_account, branch, address
--
-- Sumber: dummy/design-gold-company-atwork.jpeg (ERD)
-- Ini adalah desain BARU, terpisah dari dummy/ddl-dml.txt (lama).
-- Nama tabel sengaja beda (employer vs view_employer, dst) supaya
-- tidak bentrok dengan tabel lama - keduanya boleh hidup bareng.
--
-- Catatan hasil baca ERD:
-- - Diagram awalnya nunjukin EMPLOYER_ACCOUNT & ADDRESS link ke
--   EMPLOYER lewat cif. Diputuskan dipakein employer_id (FK formal
--   ke PK EMPLOYER) sebagai join key sebenarnya, karena cif cuma
--   business key (gak dijamin unique/enforced) - cif tetap
--   disimpen di kedua tabel sebagai kolom deskriptif aja.
-- - BRANCH: PK branch_id (surrogate) beda dengan branch_code
--   (business code yang dipakai EMPLOYER_ACCOUNT buat link ke
--   BRANCH). Sama polanya kayak employer_id vs cif di atas.
-- - branch_code (di employer_account) & postcode_id (di address)
--   dibiarin sebagai kolom deskriptif, gak dipasangin FOREIGN KEY
--   formal, karena master table branch/postcode belum tentu
--   ke-load bareng script ini (sama kayak catatan di ddl-dml.txt
--   lama).
-- - account_type di EMPLOYER_ACCOUNT nyimpen kode jenis akun
--   (mis. 'GIRO', 'DEPOSITO'); label "(is_giro)" di diagram
--   berarti kolom ini yang dipakai buat nentuin giro atau bukan.
-- - snapshot_date ditambahin di semua tabel (gak ada eksplisit di
--   diagram) biar konsisten sama pola gold layer yang lain +
--   gampang di-partition/dedup.
-- - Address di desain ini NORMALIZED: alamat primary & sekunder
--   employer sama-sama masuk tabel address, dibedain lewat flag
--   is_primary (beda sama ddl-dml.txt lama yang nyimpen alamat
--   utama flat di kolom primary_address_* langsung di employer).
-- ============================================================

CREATE SCHEMA IF NOT EXISTS `lov_gold`;

-- ------------------------------------------------------------
-- 1. employer
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer` (
  employer_id        STRING  NOT NULL,
  group_id           STRING,
  nob_id             STRING,
  cif                STRING  NOT NULL,
  employer_code      STRING,
  employer_name      STRING  NOT NULL,
  primary_telephone  STRING,
  primary_email      STRING,
  tiering            STRING,
  is_pks             BOOL,
  is_ntb             BOOL,
  snapshot_date      DATE    NOT NULL,
  PRIMARY KEY (employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 2. branch
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.branch` (
  branch_id          STRING  NOT NULL,
  branch_code        STRING,
  branch_name        STRING,
  branch_address1    STRING,
  branch_address2    STRING,
  branch_telephone   STRING,
  branch_email       STRING,
  snapshot_date      DATE    NOT NULL,
  PRIMARY KEY (branch_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 3. employer_account
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.employer_account` (
  account_id      STRING  NOT NULL,
  employer_id     STRING  NOT NULL,
  cif             STRING,
  branch_code     STRING,
  account_number  STRING,
  currency        STRING,
  product_code    STRING,
  account_type    STRING,   -- is_giro: 'GIRO' | 'DEPOSITO' | dst.
  is_sharia       BOOL,
  snapshot_date   DATE    NOT NULL,
  PRIMARY KEY (account_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- ------------------------------------------------------------
-- 4. address
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.address` (
  address_id      STRING  NOT NULL,
  postcode_id     STRING,
  employer_id     STRING  NOT NULL,
  cif             STRING,
  is_primary      BOOL,
  rt              STRING,
  rw              STRING,
  address_detail  STRING,
  subdistrict     STRING,
  district        STRING,
  city            STRING,
  province        STRING,
  snapshot_date   DATE    NOT NULL,
  PRIMARY KEY (address_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `lov_gold.employer`(employer_id) NOT ENFORCED
);

-- 1. Orphan check employer_id (harusnya 0 row)
SELECT 'employer_account' AS tabel, a.employer_id
FROM `lov_gold.employer_account` a
LEFT JOIN `lov_gold.employer` e ON a.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'address', ad.employer_id
FROM `lov_gold.address` ad
LEFT JOIN `lov_gold.employer` e ON ad.employer_id = e.employer_id
WHERE e.employer_id IS NULL;

-- 2. Setiap employer harus punya tepat 1 alamat primary
SELECT employer_id, COUNT(*) AS primary_count
FROM `lov_gold.address`
WHERE is_primary = TRUE
GROUP BY employer_id
HAVING COUNT(*) != 1;

-- 3. Contoh query self-service tanpa join
SELECT employer_name, tiering, is_pks, is_ntb, primary_email
FROM `lov_gold.employer`;

SELECT branch_name, branch_code, branch_telephone
FROM `lov_gold.branch`;
