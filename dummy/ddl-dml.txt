-- ============================================================
-- EMPLOYER GOLD (final) - BigQuery Standard SQL
-- Dataset: gold
-- 4 tabel: view_employer, view_account_branch,
--          view_employer_ownership_profile, view_employer_address
--
-- Catatan:
-- - cif & branch_code di view_account_branch, postcode_id di
--   view_employer_address dibiarin sebagai kolom deskriptif
--   (gak dipasangin FOREIGN KEY constraint) karena master
--   table-nya (branch, postcode) ada di script terpisah / belum
--   tentu ke-load bareng script ini. employer_id tetap di-FK
--   formal karena tabel induknya (view_employer) ada di script
--   yang sama.
-- - Ketemu 1 inkonsistensi kecil pas review: view_employer &
--   view_employer_address motong RT/RW jadi 2 kolom terpisah
--   (rt, rw), sedangkan view_account_branch masih 1 kolom
--   gabungan (primary_address_rt_rw). Aku ikutin persis sesuai
--   diagram masing-masing, tapi worth dirapihin konsisten nanti.
-- ============================================================
 
CREATE SCHEMA IF NOT EXISTS `gold`;
 
-- ------------------------------------------------------------
-- 1. view_employer
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `gold.view_employer` (
  employer_id                  STRING  NOT NULL,
  group_id                     STRING,
  nob_id                       STRING,
  cif                          STRING  NOT NULL,
  employer_name                STRING  NOT NULL,
  employer_code                STRING,
  stock_code                   STRING,
  ipo_date                     DATE,
  sector                       STRING,
  sub_sector                   STRING,
  industry                     STRING,
  subindustry                  STRING,
  kbli                         STRING,
  is_top_1000                  BOOL,
  is_pks                       BOOL,
  tier                         STRING,
  group_name                   STRING,
  primary_email                STRING,
  primary_contact_no           STRING,
  cl_account                   STRING,
  cl_loan_type                 STRING,
  shareholder_count            INT64,
  top_shareholder_name         STRING,
  top_shareholder_percentage   NUMERIC,
  account_count                INT64,
  primary_address_postcode     STRING,
  primary_address_rt           STRING,
  primary_address_rw           STRING,
  primary_address_detail       STRING,
  primary_address_subdistrict  STRING,
  primary_address_district     STRING,
  primary_address_city         STRING,
  primary_address_province     STRING,
  snapshot_date                DATE    NOT NULL,
  PRIMARY KEY (employer_id) NOT ENFORCED
);
 
-- ------------------------------------------------------------
-- 2. view_account_branch
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `gold.view_account_branch` (
  account_id                   STRING  NOT NULL,
  cif                          STRING,
  branch_code                  STRING,
  employer_name                STRING,
  group_name                   STRING,
  account_number                STRING,
  currency                     STRING,
  product_code                 STRING,
  account_type                 STRING,
  creation_bank_no             STRING,
  is_sharia                    BOOL,
  branch_name                  STRING,
  branch_address1              STRING,
  branch_address2              STRING,
  branch_telephone             STRING,
  branch_email                 STRING,
  primary_address_postcode     STRING,
  primary_address_rt_rw        STRING,
  primary_address_detail       STRING,
  primary_address_subdistrict  STRING,
  primary_address_district     STRING,
  primary_address_city         STRING,
  primary_address_province     STRING,
  snapshot_date                DATE    NOT NULL,
  PRIMARY KEY (account_id) NOT ENFORCED
);
 
-- ------------------------------------------------------------
-- 3. view_employer_ownership_profile
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `gold.view_employer_ownership_profile` (
  stakeholder_id         STRING  NOT NULL,
  employer_id            STRING  NOT NULL,
  employer_name          STRING,
  stock_code             STRING,
  group_name             STRING,
  sector                 STRING,
  shareholder_type       STRING,
  stakeholder_name       STRING,
  ownership_amount       NUMERIC,
  ownership_percentage   NUMERIC,
  designation            STRING,
  snapshot_date          DATE    NOT NULL,
  PRIMARY KEY (stakeholder_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `gold.view_employer`(employer_id) NOT ENFORCED
);
 
-- ------------------------------------------------------------
-- 4. view_employer_address
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `gold.view_employer_address` (
  address_id      STRING  NOT NULL,
  postcode_id     STRING,
  employer_id     STRING  NOT NULL,
  employer_name   STRING,
  group_name      STRING,
  address_type    STRING,
  rt              STRING,
  rw              STRING,
  address_detail  STRING,
  subdistrict     STRING,
  district        STRING,
  city            STRING,
  province        STRING,
  snapshot_date   DATE    NOT NULL,
  PRIMARY KEY (address_id) NOT ENFORCED,
  FOREIGN KEY (employer_id) REFERENCES `gold.view_employer`(employer_id) NOT ENFORCED
);

-- 1. Orphan check employer_id (harusnya 0 row)
SELECT 'view_employer_ownership_profile' AS tabel, o.employer_id
FROM `gold.view_employer_ownership_profile` o
LEFT JOIN `gold.view_employer` e ON o.employer_id = e.employer_id
WHERE e.employer_id IS NULL
UNION ALL
SELECT 'view_employer_address', a.employer_id
FROM `gold.view_employer_address` a
LEFT JOIN `gold.view_employer` e ON a.employer_id = e.employer_id
WHERE e.employer_id IS NULL;
 
-- 2. Cek shareholder_count / account_count di view_employer match jumlah row aktual
SELECT
  e.employer_id,
  e.shareholder_count AS shareholder_count_flat,
  COUNT(DISTINCT o.stakeholder_id) AS shareholder_count_actual,
  e.account_count AS account_count_flat,
  COUNT(DISTINCT a.account_id) AS account_count_actual
FROM `gold.view_employer` e
LEFT JOIN `gold.view_employer_ownership_profile` o ON e.employer_id = o.employer_id
LEFT JOIN `gold.view_account_branch` a ON e.cif = a.cif
GROUP BY e.employer_id, e.shareholder_count, e.account_count
ORDER BY e.employer_id;
 
-- 3. Contoh query self-service tanpa join (masing-masing tabel berdiri sendiri)
SELECT employer_name, tier, primary_address_city, primary_email, top_shareholder_name
FROM `gold.view_employer`
WHERE is_top_1000 = TRUE;
 
SELECT employer_name, branch_name, account_number, account_type
FROM `gold.view_account_branch`
