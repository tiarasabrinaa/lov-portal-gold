-- - risk_level cuma 2 kemungkinan isi: 'L' (Low) atau 'H' (High).
-- - business_date = tanggal data ini berlaku/dilaporkan, update_date
--   = kapan baris ini terakhir di-update - dua-duanya bisa beda.
-- - app/api/endpoints/nature_of_business.py punya 2 endpoint:
--     * GET /nature-of-business/          -> get all
--     * GET /nature-of-business/{nob_id}   -> get by id (list, 0/1
--       baris karena nob_id sekarang unik)
--   Filter ?source_system= yang dulu ada di endpoint kedua DIHAPUS
--   karena kolomnya udah gak ada.
-- - nob_id diisi NOB01..NOB20 - ini SEMUA nob_id yang nyata dipakai di
--   `lov_gold.gold_employer_profile` (800rb+ baris riil, dicek lewat
--   GROUP BY nob_id, COUNT(*)). corp_short_desc per nob_id disamain ke
--   `industry` yang paling dominan buat nob_id itu di data riil
--   (GROUP BY nob_id, industry ORDER BY COUNT(*) DESC), biar join dari
--   company/employer endpoints ke nob ga menghasilkan NULL. int_indcode
--   pakai kode ala-KBLI yang cocok sama industri masing-masing (dummy,
--   bukan hasil lookup KBLI resmi).
-- ============================================================

CREATE SCHEMA IF NOT EXISTS `lov_gold`;

-- ------------------------------------------------------------
-- 1. nob
-- ------------------------------------------------------------
CREATE OR REPLACE TABLE `lov_gold.nob` (
  nob_id            STRING  NOT NULL,
  business_date     DATE,
  int_indcode       STRING,
  corp_short_desc   STRING,
  corp_desc         STRING,
  risk_level        STRING,  -- cuma 'L' (Low) atau 'H' (High)
  update_date       DATE,
  PRIMARY KEY (nob_id) NOT ENFORCED
);

-- 1. nob_id harus unik (harusnya semua count = 1)
SELECT nob_id, COUNT(*) AS total
FROM `lov_gold.nob`
GROUP BY nob_id
ORDER BY nob_id;

-- 2. risk_level cuma boleh 'L' atau 'H' (harusnya 0 row)
SELECT nob_id, risk_level
FROM `lov_gold.nob`
WHERE risk_level NOT IN ('L', 'H');

-- 3. Orphan check ke gold_employer_profile.nob_id (kalau
--    ddl-dml-company-atwork-v3.txt udah di-load duluan; harusnya
--    0 row karena NOB01..NOB05 dipakai di sana)
SELECT e.nob_id
FROM `lov_gold.gold_employer_profile` e
LEFT JOIN `lov_gold.nob` n ON e.nob_id = n.nob_id
WHERE n.nob_id IS NULL;

-- 4. Simulasi GET /nature-of-business/ (get all)
SELECT nob_id, business_date, int_indcode, corp_short_desc, corp_desc, risk_level, update_date
FROM `lov_gold.nob`
ORDER BY nob_id;

-- 5. Simulasi GET /nature-of-business/NOB01 (get by id -> 1 baris)
SELECT nob_id, business_date, int_indcode, corp_short_desc, corp_desc, risk_level, update_date
FROM `lov_gold.nob`
WHERE nob_id = 'NOB01';

-- 6. Simulasi employer join (dipakai company endpoints buat narik
--    corp_short_desc & corp_desc bareng data employer)
SELECT e.employer_id, e.employer_name, e.nob_id, n.corp_short_desc, n.corp_desc
FROM `lov_gold.gold_employer_profile` e
LEFT JOIN `lov_gold.nob` n ON e.nob_id = n.nob_id
ORDER BY e.employer_id;
